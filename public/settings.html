'use strict'; const test = require('node:test'); const assert = require('node:assert'); const fs = require('node:fs'); const os = require('node:os'); const path = require('node:path'); const { patchFeatureSection, stripComments } = require('../src/config-edit'); function tmpRoot(jsonc) { const root = fs.mkdtempSync(path.join(os.tmpdir(), 'cfg-edit-')); fs.writeFileSync(path.join(root, 'config.json'), jsonc, { mode: 0o600 }); return root; } const BASE = [ '{', ' // 顶层注释不能丢', ' "active": "deepseek",', ' "features": {', ' // bili 注释也不能丢', ' "bili": {', ' // 是否启用', ' "enabled": true,', ' "intervalMinutes": 180, // 行内注释保留', ' "activeWindows": ["12:00-14:00", "18:00-09:00"],', ' "interact": {', ' "enabled": true,', ' "maxLikes": 10', ' }', ' }', ' }', '}', ].join('\n'); test('改顶层 key：注释全保留，其余值不动', function () { const root = tmpRoot(BASE); const result = patchFeatureSection(root, 'bili', { intervalMinutes: 90 }); assert.equal(result.ok, true); const text = fs.readFileSync(path.join(root, 'config.json'), 'utf8'); assert.match(text, /\/\/ 顶层注释不能丢/); assert.match(text, /\/\/ bili 注释也不能丢/); assert.match(text, /"intervalMinutes": 90, \/\/ 行内注释保留/); assert.equal(result.applied.intervalMinutes, 90); assert.equal(result.applied.activeWindows[0], '12:00-14:00'); }); test('嵌套 key（interact.maxLikes）与布尔值', function () { const root = tmpRoot(BASE); const result = patchFeatureSection(root, 'bili', { 'interact.maxLikes': 3, weekendAllDay: false }); assert.equal(result.ok, true); assert.equal(result.applied.interact.maxLikes, 3); assert.equal(result.applied.weekendAllDay, false); const text = fs.readFileSync(path.join(root, 'config.json'), 'utf8'); assert.match(text, /"maxLikes": 3/); assert.match(text, /\/\/ 是否启用/); }); test('段里缺的 key 会追加，且 JSON 仍合法', function () { const root = tmpRoot(BASE); const result = patchFeatureSection(root, 'bili', { 'interact.maxReplies': 5, owner: '主人' }); assert.equal(result.ok, true); assert.equal(result.applied.interact.maxReplies, 5); assert.equal(result.applied.owner, '主人'); // 追加后整段还能被标准 parse（尾逗号正确） const parsed = JSON.parse(stripComments(fs.readFileSync(path.join(root, 'config.json'), 'utf8'))); assert.equal(parsed.features.bili.enabled, true); }); test('段缺失时自动创建最小段', function () { const root = tmpRoot('{\n "features": {\n "mood": {\n "enabled": false\n }\n }\n}'); const result = patchFeatureSection(root, 'speakUp', { enabled: true, probability: 0.2 }); assert.equal(result.ok, true); assert.equal(result.applied.enabled, true); assert.equal(result.applied.probability, 0.2); // 原有 mood 段不受影响 assert.equal(result.applied_mood_check, undefined); const parsed = JSON.parse(stripComments(fs.readFileSync(path.join(root, 'config.json'), 'utf8'))); assert.equal(parsed.features.mood.enabled, false); }); test('数组值整体替换', function () { const root = tmpRoot(BASE); const result = patchFeatureSection(root, 'bili', { activeWindows: ['09:00-12:00'] }); assert.equal(result.ok, true); assert.deepEqual(result.applied.activeWindows, ['09:00-12:00']); }); test('写回会产生非法 JSON 时拒绝落盘', function () { // 模拟方式：patch 里带无法序列化的内容不太可能，这里直接验证 API 正常路径下文件保持可解析 const root = tmpRoot(BASE); const result = patchFeatureSection(root, 'bili', { dailyQuota: 2 }); assert.equal(result.ok, true); const text = fs.readFileSync(path.join(root, 'config.json'), 'utf8'); assert.doesNotThrow(function () { JSON.parse(stripComments(text)); }); });
